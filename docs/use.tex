% --- UC-01 ---
\subsection*{Use Case 1: Upload Academic Resource}
\begin{longtable}{|p{4cm}|p{11cm}|}
\hline
\textbf{Use Case ID:} & UC-01 \\
\hline
\textbf{Use Case Name:} & Upload Academic Resource \\
\hline
\textbf{Created By:} & Code Club \\
\textbf{Date Created:} &February 13, 2026 \\
\hline
\textbf{Actors:} & Student \\
\hline
 \textbf{Description:} & A student uploads a digital academic resource (lecture notes, project report, or question paper) to a specific subject repository for sharing with peers. \\
\hline
\textbf{Preconditions:} & 
1. Student is logged into NexusNotes. \newline
2. Student has valid "Contributor" permissions. \\
\hline
\textbf{Postconditions:} & 
1. Resource is stored in the database. \newline
2.  Resource is indexed for search and categorization. \\
\hline
\textbf{Normal Flow:} & 
\textbf{1.0 Upload Resource} \newline
1. Student navigates to the "Contribute" section. \newline
2. System displays the upload form. \newline
3. Student selects the Category (Lecture Notes, Question Bank, Project, Emerging Tech). \newline
4. Student uploads the file (PDF, DOCX, PPT). \newline
5. Student adds metadata (Subject, Topic, Tags). \newline
6. Student submits the resource. \newline
7. System validates file format and size. \newline
8. System confirms successful upload and publishes content to the repository. \\
\hline
\textbf{Alternative Flows:} & 
\textbf{UC-01.AC.1 Bulk Upload} \newline
1. Student selects "Bulk Upload" at step 1. \newline
2. Student drags and drops multiple files. \newline
3. System applies common metadata tags to all files. \\
\hline
\textbf{Exceptions:} & 
\textbf{UC-01.EX.1 Invalid File Format} \newline
1. System detects unsupported file type. \newline
2. System displays error message and requests valid format. \\
\hline
 \textbf{Priority:} & High  \\
\hline
\textbf{Business Rules:} & Maximum file size is 50MB per resource. \\
\hline
\end{longtable}

% --- UC-02 ---
\subsection*{Use Case 2: Search and Download Resource}
\begin{longtable}{|p{4cm}|p{11cm}|}
\hline
\textbf{Use Case ID:} & UC-02 \\
\hline
\textbf{Use Case Name:} & Search and Download Resource \\
\hline
\textbf{Created By:} & Code Club \\
\textbf{Date Created:} &February 13, 2026 \\
\hline
\textbf{Actors:} & Student \\
\hline
 \textbf{Description:} & Students search for specific academic materials using keywords, filters (Subject, Module), or tags, and download them for study. \\
\hline
\textbf{Preconditions:} & Student is logged into NexusNotes. \\
\hline
\textbf{Postconditions:} & File is downloaded to the student's local device. \\
\hline
\textbf{Normal Flow:} & 
\textbf{2.0 Standard Search} \newline
1. Student enters keywords in the global search bar. \newline
2. System displays results ranked by relevance and user rating. \newline
3. Student applies filters (e.g., "Computer Science", "Semester 1"). \newline
4. System updates result list. \newline
5. Student clicks on a resource to view details. \newline
6. Student clicks "Download". \newline
7. System increments the download count. \\
\hline
\textbf{Alternative Flows:} & 
\textbf{UC-02.AC.1 Browse by Repository} \newline
1. Student navigates via menu: Repository $\rightarrow$ Year $\rightarrow$ Subject. \newline
2.  System lists all resources in that category. \\
\hline
\textbf{Exceptions:} & 
\textbf{UC-02.EX.1 No Results} \newline
1. System finds no matches. \newline
2. System suggests related topics or broader terms. \\
\hline
 \textbf{Priority:} & Medium  \\
\hline
\end{longtable}
